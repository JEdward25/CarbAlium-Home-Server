2. COMANDOS DE INICIALIZAÇÃO DO COPYPARTY (TERMUX)
------------------------------------------------------------------------

# [Versão 1] Inicialização básica com loop de autorrecuperação (sem senha):
while true; do copyparty -v /caminho/da/sua/pasta:/:rw --dav-auth; sleep 2; done

# [Versão 2] Usuário único protegendo apenas a subpasta /Cofre (execução direta):
copyparty -a seu_usuario:sua_senha -v /sdcard/CarbAlium:/:rwmd -v /sdcard/CarbAlium/Cofre:/Cofre:rwmd,seu_usuario --dav-auth

# [Versão 3] Usuário único protegendo /Cofre com loop de autorrecuperação:
while true; do copyparty -a seu_usuario:sua_senha -v /sdcard/CarbAlium:/:rwmd -v /sdcard/CarbAlium/Cofre:/Cofre:rwmd,seu_usuario --dav-auth; sleep 2; done

# [Versão 4] Nome fixo do servidor (--name CarbAlium) e ajuste de minúsculas (/cofre):
copyparty --name CarbAlium -a seu_usuario:sua_senha -v /sdcard/CarbAlium:/:rwmd -v /sdcard/CarbAlium/cofre:/cofre:rwmd,seu_usuario --dav-auth

# [Versão 5 - FINAL] Dois níveis de acesso (admin e cofre) com senhas distintas (execução direta):
copyparty --name CarbAlium -a admin:senha_admin -a cofre:senha_cofre -v /sdcard/CarbAlium:/:rwmd,admin -v /sdcard/CarbAlium/cofre:/cofre:rwmd,admin,cofre --dav-auth

# [Versão 5 - FINAL COM LOOP] Comando completo de produção com reinício automático:
while true; do copyparty --name CarbAlium -a admin:senha_admin -a cofre:senha_cofre -v /sdcard/CarbAlium:/:rwmd,admin -v /sdcard/CarbAlium/cofre:/cofre:rwmd,admin,cofre --dav-auth; sleep 2; done

# [Observação - Rota Alternativa] Para forçar caixa de senha imediata sem usar /?login:
# Substituir "-v /sdcard/CarbAlium:/:rwmd,admin" por "-v /sdcard/CarbAlium:/geral:rwmd,admin"

4. COMANDOS DE DIAGNÓSTICO E MANUTENÇÃO NO TERMUX
------------------------------------------------------------------------

# Verificar histórico da sessão atual na RAM e salvar em disco:
history | grep -E "copyparty|while|termux"
history -a

# Ler histórico salvo e arquivos de inicialização:
cat ~/.bash_history | grep -E "copyparty|while|termux"
cat ~/.bashrc ~/.profile 2>/dev/null
cat ~/*.sh ~/.termux/boot/* 2>/dev/null
cat ~/.config/copyparty/*.conf ~/.termux/termux.properties 2>/dev/null

# Verificar processo ativo em tempo real na memória:
ps -ef | grep copyparty
for p in /proc/[0-9]*; do tr '\0' ' ' < "$p/cmdline" 2>/dev/null | grep -i copyparty && echo ""; done

# Localizar logs, scripts e pastas ocultas (.hist) do servidor:
ls -la ~ && find ~ /sdcard -maxdepth 3 \( -name "*.log" -o -name "*.sh" -o -name ".hist" -o -name "*copyparty*" \) 2>/dev/null
